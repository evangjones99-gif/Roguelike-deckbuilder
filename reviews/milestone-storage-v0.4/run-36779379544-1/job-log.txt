async function boundedBody(response, limit) {
  check(Number.isSafeInteger(limit) && limit > 0 && limit <= 200 * 1024 * 1024, 'Invalid download byte bound');
  const declared = response.headers.get('content-length');
  if (declared) check(Number(declared) <= limit, 'Download Content-Length exceeds expected bytes');
  check(response.body, 'Missing download body');
  const reader = response.body.getReader();
  const chunks = []; let bytes = 0;
  try {
    for (;;) {
      const {done, value} = await reader.read();
      if (done) break;
